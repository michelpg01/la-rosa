@OptIn(ExperimentalMaterial3Api::class)
@Composable
fun AppNavigation(startDestination: String = Routes.LiveTv.route) {
    val navController = rememberNavController()
    val sharedCatalogViewModel: SharedCatalogViewModel = viewModel()
    val playerViewModel: M3uLiveTvViewModel = viewModel()

    LaunchedEffect(Unit) {
        sharedCatalogViewModel.loadData()
    }

    val drawerState = rememberDrawerState(DrawerValue.Closed)
    val scope = rememberCoroutineScope()

    val navBackStackEntry by navController.currentBackStackEntryAsState()
    val currentRoute = navBackStackEntry?.destination?.route

    val isOnMovies = currentRoute == Routes.Movies.route || currentRoute?.startsWith("movieDetail") == true || currentRoute?.startsWith("m3u_movie_detail") == true
    val isOnSeries = currentRoute == Routes.Series.route || currentRoute?.startsWith("seriesDetail") == true || currentRoute?.startsWith("m3u_series_detail") == true

    val context = LocalContext.current
    val activity = context as? android.app.Activity
    val isTvDevice = remember { context.isTv() }

    LaunchedEffect(currentRoute) {
        if (currentRoute != null) {
            if (currentRoute != Routes.LiveTv.route && currentRoute != "livePlayer") {
                PlayerHolder.liveIsBuffering = false
            }
            if (!isOnMovies && !isOnSeries) {
                VodPlayerHolder.player?.let { player -> player.stop(); player.clearMediaItems() }
                VodPlayerHolder.isBuffering = false
            }
        }
    }

    val onOpenDrawer: () -> Unit = { scope.launch { drawerState.open() } }

    val onNavigate: (String) -> Unit = { route ->
        navController.navigate(route) {
            popUpTo(navController.graph.findStartDestination().id) { saveState = true }
            launchSingleTop = true
            restoreState = true
        }
        scope.launch { drawerState.close() }
    }

    val isMainScreen = currentRoute == Routes.LiveTv.route ||
            currentRoute == Routes.Movies.route ||
            currentRoute == Routes.Series.route ||
            currentRoute == MY_LIST_ROUTE ||
            currentRoute == Routes.Profile.route

    val isFullscreen by remember { derivedStateOf { PlayerUiState.isFullscreen } }
    val isCatalogDetailOpen = M3uCatalogNavigationState.isDetailVisible

    val allowDrawerGesture = !isTvDevice && isMainScreen && !isFullscreen && !isCatalogDetailOpen

    LaunchedEffect(allowDrawerGesture) {
        if (!allowDrawerGesture && drawerState.isOpen) {
            drawerState.close()
        }
    }

    // ✅ Detectar si estamos en pantallas que deben mostrar la barra lateral en TV
    val showTvSideBar = isTvDevice && isMainScreen && !isFullscreen && !isCatalogDetailOpen

    val navHostContent: @Composable () -> Unit = {
        NavHost(
            navController = navController,
            startDestination = startDestination,
            modifier = Modifier.fillMaxSize()
        ) {
            composable("login") { LoginScreen(navController = navController) }
            composable("splash") { SplashScreen(navController = navController) }

            composable(Routes.LiveTv.route, enterTransition = { EnterTransition.None }, exitTransition = { ExitTransition.None }) {
                val catalogViewModel: LiveCatalogViewModel = viewModel()
                val channels by catalogViewModel.channels.collectAsState()
                LaunchedEffect(channels) { playerViewModel.allChannels = channels }

                LiveTvHomeScreen(
                    catalogViewModel = catalogViewModel,
                    onNavigateToPlayer = { channel ->
                        playerViewModel.requestPlayback(channel)
                        navController.navigate("livePlayer")
                        ClosePerfTracker.track("NAVIGATE_TO_PLAYER")
                    },
                    onOpenDrawer = onOpenDrawer
                )
            }

            composable("livePlayer", enterTransition = { EnterTransition.None }, exitTransition = { ExitTransition.None }) {
                SideEffect {
                    activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_SENSOR_LANDSCAPE
                }
                DisposableEffect(Unit) {
                    onDispose {
                        activity?.requestedOrientation = ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED
                        PlayerUiState.isFullscreen = false
                    }
                }
                LiveTvPlayerScreen(
                    playerViewModel = playerViewModel,
                    onBack = {
                        navController.popBackStack()
                    }
                )
            }

            composable(Routes.Movies.route) {
                MoviesScreen(navController, sharedCatalogViewModel, onOpenDrawer)
            }
            composable(Routes.Series.route) {
                SeriesScreen(navController, sharedCatalogViewModel, onOpenDrawer)
            }
            composable(MY_LIST_ROUTE) {
                CatalogScreen(
                    listName = "Mi lista",
                    contentTypeFilter = listOf(
                        M3uContentType.MOVIE,
                        M3uContentType.SERIES,
                        M3uContentType.NOVELA,
                        M3uContentType.ANIME
                    ),
                    viewModel = sharedCatalogViewModel,
                    navController = navController,
                    onBack = { navController.popBackStack() },
                    onOpenDrawer = onOpenDrawer
                )
            }
            composable(Routes.Profile.route) { ProfileScreen(navController = navController) }
        }
    }

    val sideBarFocusRequester = remember { FocusRequester() }

    if (isTvDevice) {
        // ✅ MODO TV: barra lateral fija + NavHost
        Row(
            modifier = Modifier
                .fillMaxSize()
                .background(Color(0xFF101828))
                // 📺 Tecla MENU del control -> salta directo al menú lateral
                .onPreviewKeyEvent { event ->
                    val native = event.nativeKeyEvent
                    if (native.action == android.view.KeyEvent.ACTION_DOWN &&
                        native.keyCode == android.view.KeyEvent.KEYCODE_MENU
                    ) {
                        runCatching { sideBarFocusRequester.requestFocus() }
                        true
                    } else {
                        false
                    }
                }
        ) {
            // 1. La barra lateral ocupa su propio espacio (108.dp), con margen para el foco.
            if (showTvSideBar) {
                TvSideBar(
                    currentRoute = currentRoute,
                    onNavigate = onNavigate,
                    firstItemFocusRequester = sideBarFocusRequester
                )
            }

            // 2. El contenido ocupa el resto de la pantalla
            Box(modifier = Modifier.weight(1f).fillMaxHeight()) {
                navHostContent()
            }
        }
    } else {
        // ✅ MODO CELULAR: drawer normal con swipe
        ModalNavigationDrawer(
            drawerState = drawerState,
            gesturesEnabled = allowDrawerGesture,
            drawerContent = {
                AppDrawerContent(currentRoute = currentRoute, onNavigate = onNavigate)
            },
            content = { navHostContent() }
        )
    }
}

@Composable
private fun TvSideBarItem(
    icon: ImageVector,
    label: String,
    selected: Boolean,
    selectedColor: Color,
    contentColor: Color,
    onClick: () -> Unit,
    modifier: Modifier = Modifier
) {
    var isFocused by remember { mutableStateOf(false) }
    val interactionSource = remember { MutableInteractionSource() }
    val itemShape = RoundedCornerShape(12.dp)
    val scale by animateFloatAsState(
        targetValue = if (isFocused) 1.045f else 1f,
        animationSpec = tween(durationMillis = 220, easing = FastOutSlowInEasing),
        label = "sideBarScale"
    )
    val tint = when {
        isFocused -> Color.White
        selected -> selectedColor
        else -> contentColor
    }

    Column(
        modifier = modifier
            .fillMaxWidth()
            .padding(horizontal = 5.dp, vertical = 3.dp)
            .graphicsLayer { scaleX = scale; scaleY = scale }
            .zIndex(if (isFocused) 1f else 0f)
            .clip(itemShape)
            .background(
                when {
                    isFocused -> Color.White.copy(alpha = 0.16f)
                    selected -> selectedColor.copy(alpha = 0.12f)
                    else -> Color.Transparent
                }
            )
            // ✅ ANTES del clickable y con hasFocus
            .onFocusChanged { isFocused = it.hasFocus }
            // ✅ UN solo foco, y este SÍ tiene onClick -> OK funciona
            .clickable(
                interactionSource = interactionSource,
                indication = null,
                onClick = onClick
            )
            .padding(vertical = 8.dp),
        horizontalAlignment = Alignment.CenterHorizontally
    ) {
        Icon(
            imageVector = icon,
            contentDescription = label,
            tint = tint,
            modifier = Modifier.size(28.dp)
        )
        Spacer(modifier = Modifier.height(4.dp))
        Text(
            text = label,
            color = tint,
            fontSize = 11.sp,
            fontWeight = if (selected || isFocused) FontWeight.Bold else FontWeight.Medium,
            maxLines = 1,
            softWrap = false
        )
    }
}

@Composable
private fun Modifier.tvFocusZoom(isTvDevice: Boolean): Modifier {
    var isFocused by remember { mutableStateOf(false) }
    val scale by animateFloatAsState(
        targetValue = if (isTvDevice && isFocused) 1.02f else 1f,
        animationSpec = tween(durationMillis = 220, easing = FastOutSlowInEasing),
        label = "catalogControlFocusZoom"
    )
    return this
        .onFocusChanged { isFocused = it.isFocused }
        .graphicsLayer { scaleX = scale; scaleY = scale }
        .zIndex(if (isFocused) 1f else 0f)
}

@OptIn(ExperimentalFoundationApi::class)
@Composable
private fun UnifiedPosterGrid(
    items: List<GridItemData>,
    isTvDevice: Boolean,
    firstItemFocusRequester: FocusRequester? = null,
    focusRequesterForItem: ((GridItemData) -> FocusRequester)? = null,
    onPosterFocused: ((FocusRequester) -> Unit)? = null
) {
    LazyVerticalGrid(
        columns = GridCells.Adaptive(minSize = 150.dp),
        modifier = Modifier.fillMaxSize(),
        contentPadding = PaddingValues(horizontal = 12.dp, vertical = 12.dp),
        horizontalArrangement = Arrangement.spacedBy(10.dp),
        verticalArrangement = Arrangement.spacedBy(16.dp)
    ) {
        itemsIndexed(
            items = items,
            key = { index, item -> "poster_${index}_${item.title}_${item.yearFallback.orEmpty()}_${item.logo}" }
        ) { index, item ->
            var posterHasFocus by remember(item.title, item.logo) { mutableStateOf(false) }
            val cardFocusRequester = when {
                index == 0 && firstItemFocusRequester != null -> firstItemFocusRequester
                focusRequesterForItem != null -> focusRequesterForItem(item)
                else -> null
            }
            Box(
                modifier = Modifier.fillMaxWidth().zIndex(if (posterHasFocus) 1f else 0f),
                contentAlignment = Alignment.TopCenter
            ) {
                FocusZoomPosterCard(
                    item = item,
                    isTvDevice = isTvDevice,
                    modifier = Modifier.width(if (isTvDevice) 135.dp else 150.dp).then(
                        if (cardFocusRequester != null) Modifier.focusRequester(cardFocusRequester) else Modifier
                    ),
                    onFocusChanged = { focused ->
                        posterHasFocus = focused
                        if (focused && cardFocusRequester != null) onPosterFocused?.invoke(cardFocusRequester)
                    }
                )
            }
        }
    }
}

@Composable
private fun FocusZoomPosterCard(
    item: GridItemData,
    isTvDevice: Boolean,
    modifier: Modifier = Modifier,
    onFocusChanged: (Boolean) -> Unit = {}
) {
    var isFocused by remember(item.title, item.logo) { mutableStateOf(false) }

    NetflixPosterCard(
        item = item,
        modifier = modifier
            .width(if (isTvDevice) 135.dp else 150.dp)
            .onFocusChanged {
                isFocused = it.isFocused
                onFocusChanged(it.isFocused)
            }
            // En TV indicamos el foco con un borde, sin agrandar el póster.
            .then(
                if (isTvDevice && isFocused) {
                    Modifier.border(3.dp, Color.White, RoundedCornerShape(8.dp))
                } else {
                    Modifier
                }
            )
            .zIndex(if (isFocused) 1f else 0f)
    )
}

    }.asReversed()

        when (it) {

    fun rowFocusModifier(rowKey: String, ch: M3uChannel, rowItems: List<M3uChannel>): Modifier {
        if (rowItems.isEmpty()) return Modifier
        val idx = rowOrder.indexOf(rowKey)
        val isFirst = ch.url == rowItems.first().url
        val isLast = ch.url == rowItems.last().url
        val prevKey = rowOrder.getOrNull(idx - 1)
        val nextKey = rowOrder.getOrNull(idx + 1)
        val baseIndex = (if (topCarousel.isNotEmpty()) 1 else 0) + (if (filteredRecentItems.isNotEmpty()) 1 else 0)
        val requesterKey = "$rowKey|${ch.url}"
        val cardRequester = rowPosterFocusRequesters.getOrPut(requesterKey) {
            if (isFirst) firstPosterRequesters[rowKey]!! else FocusRequester()
        }
        return Modifier
            .focusRequester(cardRequester)
            .onFocusChanged { if (it.isFocused) lastFocusedPosterRequester = cardRequester }
            .focusProperties {
                if (isLast) {
                    // Desde el último póster, derecha lleva a "Ver más".
                    right = verMasRequesters[rowKey]!!

                    // Conserva también el acceso hacia arriba al encabezado.
                    up = verMasRequesters[rowKey]!!
                }
            }
            .onKeyEvent { event ->
                if (event.type != KeyEventType.KeyDown) return@onKeyEvent false
                val targetKey = when (event.key) {
                    Key.DirectionDown -> if (isLast) nextKey else null
                    Key.DirectionUp -> if (isFirst) prevKey else null
                    else -> null
                } ?: return@onKeyEvent false
                // Saltos entre filas: primero hacemos scroll (la fila se compone) y LUEGO pedimos foco.
                val targetIndex = baseIndex + rowOrder.indexOf(targetKey)
                catalogScope.launch {
                    listState.animateScrollToItem(targetIndex)
                    runCatching { firstPosterRequesters[targetKey]?.requestFocus() }
                }
                true
            }

    var showRefreshProgressDialog by remember { mutableStateOf(false) }

                            firstItemFocusRequester = firstPosterFocusRequester,

                            onPosterFocused = { lastFocusedPosterRequester = it }

                                Box(
                                    modifier = Modifier
                                        .align(Alignment.TopCenter)
                                        .padding(top = 16.dp)

                                            color = M3uRedBadge,
                                            strokeWidth = 2.dp

                                        Text(
                                            text = "Actualizando catálogo...",
