private fun isWifiConnected(context: Context): Boolean {
    val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager
    val network = connectivityManager.activeNetwork ?: return false
    val capabilities = connectivityManager.getNetworkCapabilities(network) ?: return false
    return capabilities.hasTransport(NetworkCapabilities.TRANSPORT_WIFI)
}

private fun extractYear(title: String): String =
    Regex("\\((\\d{4})\\)").find(title)?.groupValues?.get(1) ?: ""

private fun cleanTitle(title: String): String =
    title.replace(Regex("\\(\\d{4}\\)|\\[.*?\\]|\\{.*?\\}"), "").trim()


@Composable
fun M3uMovieDetailScreen(
    movie: M3uChannel,
    listId: String,
    tmdbInfo: TmdbMediaInfo?,
    tmdbLoading: Boolean,
    allMovies: List<M3uChannel>,

    onBack: () -> Unit,
    onMyListChanged: () -> Unit = {}
) {
    val context = LocalContext.current
    val isTvDevice = remember { context.isTv() }
    val application = context.applicationContext as Application

        factory = SharedCatalogViewModelFactory(application)
    )
    val activity = context as? Activity

    fun restoreRequestedOrientation() {
        activity?.requestedOrientation = if (isTvDevice) {
            ActivityInfo.SCREEN_ORIENTATION_UNSPECIFIED
        } else {
            ActivityInfo.SCREEN_ORIENTATION_USER_PORTRAIT
        }
    }

    val scrollState = rememberScrollState()
    val backFocusRequester = remember(movie.url) { FocusRequester() }
    val playFocusRequester = remember(movie.url) { FocusRequester() }
    val serverSelectorFocusRequester = remember(movie.url) { FocusRequester() }
    val firstSimilarPosterFocusRequester = remember(movie.url) { FocusRequester() }
    val trackState = remember { TrackState() }
    val coroutineScope = rememberCoroutineScope()

    val liveAllMovies by sharedCatalogViewModel.moviesList.collectAsState()
    val liveMovie = remember(movie.url, liveAllMovies) {
        liveAllMovies.firstOrNull { it.url == movie.url } ?: movie
    }
    var isInMyList by remember(movie.url) {
        mutableStateOf(M3uMyListStore.contains(context, M3uMyListEntry.MOVIE, movie.url))
    }
    val movieSynopsis = tmdbInfo?.overview?.takeIf { it.isNotBlank() }
        ?: liveMovie.sinopsis.orEmpty()
    var showFullSynopsis by remember(movie.url) { mutableStateOf(false) }

    LaunchedEffect(movie.url) {
        scrollState.scrollTo(0)
        WorkerResolver.getSessionToken(context)
    }

    var currentStreamUrl by remember { mutableStateOf<String?>(null) }
    var currentEpTitle by remember { mutableStateOf("") }
    var playerKey by remember { mutableStateOf<String?>(null) }

    var activePlayerType by remember { mutableIntStateOf(1) }

    var isExtractingPlay by remember { mutableStateOf(false) }
    var isExtractingCast by remember { mutableStateOf(false) }
    var showCastDialog by remember { mutableStateOf(false) }
    var castStreamUrl by remember { mutableStateOf<String?>(null) }
    var castReferer by remember { mutableStateOf<String?>(null) }
    var castCookies by remember { mutableStateOf("") }

    var currentReferer by remember { mutableStateOf<String?>(null) }
    var currentRefererFallback by remember { mutableStateOf<String?>(null) }
    var isSilentMode by remember { mutableStateOf(false) }
    var preExtractedCookies by remember { mutableStateOf("") }

    var showServerDeadOverlay by remember { mutableStateOf(false) }
    var serverDeadMessageText by remember { mutableStateOf("") }
    var showNoServersOverlay by remember { mutableStateOf(false) }

    val servers = remember(movie.url, liveMovie.servers) {
        liveMovie.servers.ifEmpty {
            listOf(M3uServer("Servidor Único", movie.url, liveMovie.isEmbed))
        }
    }

    var selectedServer by remember(movie.url, servers) { mutableStateOf(servers.first()) }
    var expandedServers by remember { mutableStateOf(false) }
    var retryCount by remember { mutableIntStateOf(0) }

    var similarMoviesByApi by remember { mutableStateOf<List<M3uChannel>>(emptyList()) }

    LaunchedEffect(movie.url, allMovies.size, allMovies.hashCode()) {
        withContext(Dispatchers.Default) {
            val snapshotMovies = allMovies
            val genre = liveMovie.genre
            val byGenre = if (!genre.isNullOrBlank()) {
                val genresList = genre.split(Regex("[,|/·]")).map { it.trim().lowercase() }
                snapshotMovies.filter {
                    it.url != movie.url &&
                            !it.genre.isNullOrBlank() &&
                            it.genre.split(Regex("[,|/·]")).any { g -> genresList.contains(g.trim().lowercase()) }
                }.shuffled()
            } else emptyList()

            val finalList = if (byGenre.size < 10) {
                val resto = snapshotMovies.filter { it.url != movie.url && !byGenre.contains(it) }.shuffled()
                (byGenre + resto).take(15)
            } else {
                byGenre.take(15)
            }

            similarMoviesByApi = finalList
        }
    }

    val historyEntry = remember(movie.url, listId) {
        M3uWatchHistoryStore.getEntry(movie.url, M3uWatchType.MOVIE)
            ?.takeIf { it.listId.isBlank() || it.listId == listId }
    }

    LaunchedEffect(movie.url, historyEntry, servers) {
        val savedServerUrl = historyEntry?.serverUrl?.takeIf { it.isNotBlank() }
        if (savedServerUrl != null) {
            selectedServer = servers.firstOrNull { it.url == savedServerUrl } ?: selectedServer
        }
    }

    val resumePositionMs = remember(historyEntry) {
        historyEntry?.takeIf { !it.isFinished && it.positionMs > 0L }?.positionMs ?: 0L
    }

    val shouldResume = resumePositionMs > 0L

    fun saveMovieAsRecent(
        positionMs: Long = 0L,
        durationMs: Long = 0L,
        server: M3uServer? = selectedServer
    ) {
        M3uWatchHistoryStore.save(
            M3uWatchEntry(
                contentUrl = movie.url,
                watchType = M3uWatchType.MOVIE,
                title = liveMovie.title,
                logoUrl = liveMovie.logo,
                listId = listId,
                positionMs = positionMs,
                durationMs = durationMs,
                serverName = server?.name.orEmpty(),
                serverUrl = server?.url.orEmpty()
            )
        )
    }

    fun openServer(server: M3uServer, saveRecent: Boolean = true, isForCast: Boolean = false) {
        currentEpTitle = liveMovie.title

        if (saveRecent && !isForCast) {
            saveMovieAsRecent(
                positionMs = historyEntry?.positionMs ?: 0L,
                durationMs = historyEntry?.durationMs ?: 0L,
                server = server
            )
        }

        coroutineScope.launch {
            if (isForCast) isExtractingCast = true else isExtractingPlay = true

            val resolvedUrl = if (WorkerResolver.isWorkerUrl(server.url)) {
                android.util.Log.d("MOVIE_DETAIL", "🔍 Resolviendo worker en película: ${server.url}")
                WorkerResolver.resolveIfWorker(server.url, isSeries = false, context = context) ?: server.url
            } else {
                server.url
            }
            android.util.Log.d("VOD_EXTRACT", "🎬 Resolviendo al click: $resolvedUrl")
            val result = VideoExtractor.resolve(context, resolvedUrl)

            if (isForCast) {
                if (result.playerType == 3) {
                    Toast.makeText(context, "Este servidor no permite Cast directo.", Toast.LENGTH_LONG).show()
                } else {
                    castStreamUrl = result.targetUrl
                    castReferer = result.referer
                    castCookies = result.cookies
                    showCastDialog = true
                }
                isExtractingCast = false
            } else {
                currentStreamUrl = result.targetUrl
                currentReferer = result.referer
                currentRefererFallback = null
                isSilentMode = result.isSilentMode
                activePlayerType = result.playerType
                preExtractedCookies = result.cookies

                castStreamUrl = result.targetUrl
                castReferer = result.referer
                castCookies = result.cookies

                CookieManager.getInstance().flush()
                playerKey = server.url
                isExtractingPlay = false
            }
        }
    }

    val forceNextServer = {
        val currentIndex = servers.indexOf(selectedServer)
        if (currentIndex < servers.size - 1) {
            android.util.Log.w("VOD_EXTRACT", "⚠️ Forzando cambio al siguiente servidor (${currentIndex + 2}/${servers.size})")
            retryCount = 0
            selectedServer = servers[currentIndex + 1]
            currentStreamUrl = null
            playerKey = null
            castStreamUrl = null
            isSilentMode = false
            VodPlayerHolder.player?.release()
            VodPlayerHolder.player = null
            CoroutineScope(Dispatchers.Main).launch {
                delay(150)
                openServer(selectedServer, saveRecent = false)
            }
        } else {
            android.util.Log.e("VOD_EXTRACT", "❌ No hay más servidores disponibles")
            showNoServersOverlay = true
            playerKey = null
            PlayerUiState.isFullscreen = false
            restoreRequestedOrientation()
        }
    }

    LaunchedEffect(movie.url) {
        playerKey = null
        currentStreamUrl = null
        currentReferer = null
        currentRefererFallback = null
        isSilentMode = false
        castStreamUrl = null
        retryCount = 0
        showServerDeadOverlay = false
        showNoServersOverlay = false
        serverDeadMessageText = ""

        VodPlayerHolder.player?.release()
        VodPlayerHolder.player = null
        PlayerUiState.isFullscreen = false

        WebStorage.getInstance().deleteAllData()
        CookieManager.getInstance().removeAllCookies(null)
        CookieManager.getInstance().flush()
    }

    val triggerServerFallback = {
        if (retryCount < 3) {
            retryCount++
            Toast.makeText(context, "Reintentando... ($retryCount/3)", Toast.LENGTH_SHORT).show()
            val tempUrl = currentStreamUrl
            currentStreamUrl = null
            playerKey = null

            CoroutineScope(Dispatchers.Main).launch {
                delay(800)
                currentStreamUrl = tempUrl
                playerKey = tempUrl
            }
        } else {
            // ✅ REPORTAR FALLO FINAL ANTES DE CAMBIAR DE SERVIDOR
            SupabaseFailReporter.reportCanalFail(context, movie.url, selectedServer.url)
            forceNextServer()
        }
    }

    val isFullscreen = PlayerUiState.isFullscreen

    BackHandler(enabled = true) {
        if (isFullscreen) {
            PlayerUiState.isFullscreen = false
            restoreRequestedOrientation()
        } else {
            onBack()
        }
    }

    Box(
        modifier = Modifier
            .fillMaxSize()
            .background(Color.Black)
    ) {
        IconButton(
            onClick = {
                if (isFullscreen) {
                    PlayerUiState.isFullscreen = false
                    restoreRequestedOrientation()
                } else {
                    onBack()
                }

                .align(Alignment.TopStart)
                .padding(16.dp)
                .zIndex(10f)
                .background(Color.Black.copy(alpha = 0.5f), CircleShape)

                Icons.AutoMirrored.Filled.ArrowBack,

                tint = Color.White,

        }

                    .fillMaxSize()
                    .verticalScroll(scrollState)

                    }
                    openServer(selectedServer, saveRecent = true)

// 📺 En TV el foco inicial cae directamente en el botón Reproducir.

                    if (playerKey == null && isTvDevice) {
                        delay(300)

                        .fillMaxWidth()

                        .border(

                            shape = RoundedCornerShape(if (isTvDevice) 16.dp else 12.dp)

                    if (playerKey == null) {

private fun isWifiConnected(context: Context): Boolean {
    val connectivityManager = context.getSystemService(Context.CONNECTIVITY_SERVICE) as ConnectivityManager

                currentReferer = finalReferer
                currentRefererFallback = null
                isSilentMode = result.isSilentMode
                activePlayerType = result.playerType
                preExtractedCookies = result.cookies

                CookieManager.getInstance().flush()
                playerKey = server.url
                isExtractingPlay = false
            }
        }
    }

    val forceNextServer = {
        val currentIndex = selectedServer?.let { servers.indexOf(it) } ?: -1
        if (currentIndex in 0 until servers.lastIndex) {
            Log.w("VOD_EXTRACT", "⚠️ Forzando cambio al siguiente servidor (${currentIndex + 2}/${servers.size})")
            retryCount = 0
            selectedServer = servers[currentIndex + 1]
            currentStreamUrl = null
            playerKey = null
            isSilentMode = false
            VodPlayerHolder.player?.release()
            VodPlayerHolder.player = null

            CoroutineScope(Dispatchers.Main).launch {
                delay(150)
                selectedEpisode?.let { ep ->
                    selectedServer?.let { srv ->
                        openEpisodeServer(ep, srv, saveRecent = false)
                    }
                }
            }
        } else {
            Log.e("VOD_EXTRACT", "❌ No hay más servidores disponibles")
            showNoServersOverlay = true
            playerKey = null
            PlayerUiState.isFullscreen = false
            restoreRequestedOrientation()
        }
    }

    val triggerServerFallback = {
        if (retryCount < 3) {
            retryCount++
            Toast.makeText(context, "Reintentando... ($retryCount/3)", Toast.LENGTH_SHORT).show()
            val tempUrl = currentStreamUrl
            currentStreamUrl = null
            playerKey = null

            scope.launch {
                delay(800)
                currentStreamUrl = tempUrl
                playerKey = tempUrl
            }
        } else {
            // ✅ REPORTAR FALLO FINAL ANTES DE CAMBIAR DE SERVIDOR
            val epUrl = selectedEpisode?.url ?: ""
            val srvUrl = selectedServer?.url ?: ""
            if (epUrl.isNotBlank() && srvUrl.isNotBlank()) {
                SupabaseFailReporter.reportCanalFail(context, epUrl, srvUrl)
            }
            forceNextServer()
        }
    }

        currentRefererFallback = null
        isSilentMode = false
        isExtractingPlay = false
        isExtractingCast = false
        showCastDialog = false

            }
            return
        }
        focusEpisodeAt(index)
    }

        currentStreamUrl = null

                                onTimeoutError = {

                                    val isIbraVariantM3u8 = isIbraStream && interceptedUrl.substringBefore("?").endsWith(".m3u8", ignoreCase = true)

                                            Log.d("SERIES_EMBED", "⏭️ Ignorando master duplicado")
                                            return@EmbedVideoPlayer

                                        preExtractedCookies = cookies
                                        isSilentMode = false

                                                    else -> currentReferer
                                                }
                                                isDoodstreamStream -> when {
                                                    selectedServer?.url?.contains("playmogo.com", ignoreCase = true) == true -> "https://playmogo.com/"
                                                    selectedServer?.url?.contains("doodstream.com", ignoreCase = true) == true -> "https://doodstream.com/"
                                                    selectedServer?.url?.contains("dood.video", ignoreCase = true) == true -> "https://dood.video/"
                                                    selectedServer?.url?.contains("ds2play.com", ignoreCase = true) == true -> "https://ds2play.com/"
                                                    currentReferer?.contains("playmogo.com", ignoreCase = true) == true -> "https://playmogo.com/"
                                                    currentReferer?.contains("doodstream.com", ignoreCase = true) == true -> "https://doodstream.com/"
                                                    else -> currentReferer
                                                }
                                                interceptedUrl.contains("dropcdn.io", ignoreCase = true) -> "https://dr0pstream.com/"
                                                interceptedUrl.contains("streamwish", ignoreCase = true) ||
                                                        interceptedUrl.contains("playnixes", ignoreCase = true) ||
                                                        interceptedUrl.contains("hgplaycdn", ignoreCase = true) -> "https://streamwish.to/"
                                                interceptedUrl.contains("callistanise", ignoreCase = true) ||
                                                        interceptedUrl.contains("affordablewebinars", ignoreCase = true) ||
                                                        interceptedUrl.contains("vidhide", ignoreCase = true) -> "https://callistanise.com/"
                                                interceptedUrl.contains("hanerix", ignoreCase = true) -> "https://hanerix.com/"
                                                interceptedUrl.contains("sagaciousslumber.site", ignoreCase = true) -> "https://cloudorchestranova.com/"
                                                interceptedUrl.contains("cloudorchestranova.com", ignoreCase = true) -> "https://vsembed.ru/"
                                                else -> currentReferer
                                            }
                                            currentRefererFallback = null
                                        }

                                        Log.d("SERIES_EMBED", "✅ Transicionando a ExoPlayer: playerType=1, referer=$currentReferer, url=$interceptedUrl")
                                        activePlayerType = 1
                                    } else {
                                        Log.w("SERIES_EMBED", "⚠️ Stream rechazado por validador: $interceptedUrl")
                                    }
                                },
                                onBackClick = {
                                    if (isFullscreen && !isTvDevice) {
                                        leaveFullscreenAndKeepPlayback()
                                    } else {
                                        stopPlaybackAndReturnToEpisodes()
                                    }
                                }
                            )
                        }
                    }

                    if (showServerDeadOverlay) {
                        Box(
                            modifier = Modifier.fillMaxSize().background(Color.Black.copy(alpha = 0.92f)).zIndex(50f),
                            contentAlignment = Alignment.Center
                        ) {
                            Column(
                                horizontalAlignment = Alignment.CenterHorizontally,
                                modifier = Modifier.padding(32.dp)
                            ) {
                                Icon(Icons.Default.Warning, contentDescription = null, tint = Color(0xFFFFC107), modifier = Modifier.size(72.dp))
                                Spacer(Modifier.height(16.dp))
                                Text(text = serverDeadMessageText, color = Color.White, fontSize = 18.sp, fontWeight = FontWeight.Bold, textAlign = TextAlign.Center)
                                Spacer(Modifier.height(24.dp))
                                CircularProgressIndicator(color = Color(0xFFE50914))
                            }
                        }
                    }
                }
            }
        }

        if (showNoServersOverlay) {
            Box(
                modifier = Modifier.fillMaxSize().background(Color.Black.copy(alpha = 0.96f)).zIndex(200f),
                contentAlignment = Alignment.Center
            ) {
                Column(
                    horizontalAlignment = Alignment.CenterHorizontally,
                    modifier = Modifier.padding(32.dp)
                ) {
                    Icon(Icons.Default.Warning, contentDescription = null, tint = Color(0xFFE50914), modifier = Modifier.size(80.dp))
                    Spacer(Modifier.height(20.dp))
                    Text(
                        text = "Servidores no disponibles",
                        color = Color.White,
                        fontSize = 22.sp,
                        fontWeight = FontWeight.Bold,
                        textAlign = TextAlign.Center
                    )
                    Spacer(Modifier.height(12.dp))
                    Text(
                        text = "Por el momento nuestros servidores no se encuentran disponibles.\nIntenta de nuevo más tarde.",
                        color = Color.White.copy(alpha = 0.75f),
                        fontSize = 15.sp,
                        lineHeight = 22.sp,
                        textAlign = TextAlign.Center
                    )
                    Spacer(Modifier.height(28.dp))
                    Button(
                        onClick = {
                            showNoServersOverlay = false
                            playerKey = null
                            currentStreamUrl = null
                            onBack()
                        },
                        colors = ButtonDefaults.buttonColors(containerColor = Color(0xFFE50914))
                    ) {
                        Text("Volver", color = Color.White, fontWeight = FontWeight.Bold)
                    }
                }
            }
        }
    }
